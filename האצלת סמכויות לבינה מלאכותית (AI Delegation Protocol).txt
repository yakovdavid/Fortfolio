        hist = ticker.history(period="5d")
        if len(hist) < 2:
            return ""
        
        prev_close = hist['Close'].iloc[-2]
        current_price = hist['Close'].iloc[-1]
        pct_change = (current_price - prev_close) / prev_close
        
        if abs(pct_change) >= threshold:

            news_data = ticker.news
            if not news_data:

            headlines = [article.get('title', 'No Title') for article in news_data[:3]]
            news_str = " | ".join(headlines)
            return f"  -> [ANOMALY: {direction} {abs(pct_change)*100:.1f}%] NEWS: {news_str}"
            
        return ""
    except Exception as e:
        return f"  -> [NEWS FETCH ERROR] Could not fetch data for {ticker_symbol}: {str(e)}"
